# 为数组分配元素
```java
//为数组分配元素  
        float[] boy;  
        boy = new float[4];//  ==  float[] boy = new float[4]  
        boy[0] = 12;  
        boy[1] = 23.00f;  
        boy[2] = 100;  
        boy[3] = 10.23f;  
        int[][] myOne = new int[3][4];  
        myOne[0][0] = 0;  
        myOne[0][1] = 1;  
        myOne[0][2] = 2;  
        myOne[0][3] = 3;  
        myOne[1][0] = 10;  
        myOne[2][3] = 23;  
        int[][] myTwo = new int[3][];  
        myTwo[0] = new int[6];  
        myTwo[1] = new int[12];  
        myTwo[2] = new int[8];  
        myTwo[0][5] = 5; 
```

# 数组的引用
```java
//数组引用  
        int arr1[] = {1,2,3},arr2[] = {4,5};  
        System.out.println("arr1的个数:" + arr1.length);  
        System.out.println("arr2的个数:" + arr2.length);  
        int address = System.identityHashCode(arr1);  
        System.out.println("arr1的地址：" + address);  
        address = System.identityHashCode(arr2);  
        System.out.println("arr2的地址：" + address);  
        arr1 = arr2;//将arr1指向了arr2的地址  
        System.out.println("arr1 = arr2后---------------");  
        System.out.println("arr1的个数:" + arr1.length);  
        System.out.println("arr2的个数:" + arr2.length);  
        address = System.identityHashCode(arr1);  
        System.out.println("arr1的地址：" + address);  
        address = System.identityHashCode(arr2);  
        System.out.println("arr2的地址：" + address);  
  
        //结果  
        /*4  
        arr1的个数:3  
        arr2的个数:2  
        arr1的地址：2055281021  
        arr2的地址：1554547125  
        arr1 = arr2后---------------  
        arr1的个数:2  
        arr2的个数:2  
        arr1的地址：1554547125  
        arr2的地址：1554547125*/  
```
