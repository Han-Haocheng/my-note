顺序栈：用连续的内存空间存储栈
```c
#define MAXSIZE 100
typedef struct
{ 
	SElemType *base;
	SElemType *top;
	int stacksize;
}SqStack;
```


### 5.1 栈初始化

```C++
Status InitStack(SqStack&S)
{
	S.base = (SElemType*)malloc(MAXSIZE*sizeof(SElemType));
	if(!S.base) exit(OVERFLOW); // 内存分配失败
	S.top = S.base;
	S.stacksize = MAXSIZE;
	return OK;
};
```

链栈

```c
Status InitStack(LinkStack&S)
{
	S=NULL;
	return OK;
}
```

### 5.2 栈判空

```c++
Status StackEmpty(SqStack S){
	return S.top == S.base;
}

Status StackEmpty(LinkStack &S){
	return S == NULL;
}
```

### 5.3 栈获取长度

```c++
int StackLength(SqStack S){
	return S.top - S.base;
}
```


### 5.4 栈清空

```c++
Status ClearStack(SqStack S){
	if(S.base) S.top = S.base;
	return OK;
}
```

### 5.5 栈销毁

```c++
Status DestroyStack(SqStack &S){
	if(!S.base)return OK;
	delete S.base;
	S.stacksize = 0;
	S.base = S.top = NULL;
	return OK;
}
```

### 5.7 栈出栈

顺序栈出栈
```C++
Status Pop(SqStack &S, SElemType &e){
	if(S.top == S.base)return ERROR;
	e = *(--S.top);
	return OK;
}
```
