```c++
// 函数功能：用Bresenham算法画一条从p0(100,50)到p1(200,150)的黑色直线 
void BresenhamLine(CDC* pDC) { 
// 1. 定义直线的两个端点：起点p0，终点p1，临时点p（用来遍历画像素） 
	CPoint p0(100, 50), p1(200, 150), p; 
// 2. 计算x方向、y方向的总步长（差值） 
	int dx, dy; 
	dx = p1.x - p0.x;   // dx = 200-100 = 100 → x轴要走100步 
	dy = p1.y - p0.y;               // dy = 150-50 = 100 → y轴要走100步 
// 3. 定义斜率k，决策参数d（Bresenham核心） 
	double k, d; 
	k = (double)dy / dx;   // 计算直线斜率 k = 100/100 = 1 
	d = 0.5 - k;           //d = 0.5 - 1 = -0.5  初始化中点误差项
// 4. 核心循环：从起点x开始，一步步向右画，直到终点x 
 // p = p0：从起点开始画 
 // p.x < p1.x：x没到终点就一直循环（不画终点像素） 
 // p.x++：每循环一次，x向右移动1个像素 
	for (p = p0; p.x < p1.x; p.x++) { 
    // 点亮当前坐标p位置的像素，颜色：纯黑(RGB0,0,0) 
		pDC->SetPixelV(p, RGB(0, 0, 0)); 
// 5. Bresenham核心判断：决定下一步y要不要+1 
		if (d < 0)       // 决策参数d < 0，需要向上走一步 
		{   p.y++;       // y坐标+1，点亮上方的像素 
			d += 1 - k; // 更新决策参数 
		} 
		else // d ≥ 0，保持当前y不变，不向上走 
		{   d -= k; 
		// 只更新决策参数 
		} 
	} 
}
```
其中：
1. windows的屏幕起点坐标为(0,0)
2. d = 0.5 - k ：	
	1. d = 0.5（分界线） - k（直线上升量）
	2. k的取值：
		1. 0~π/4 :  
			1. k = [0,1)
		2. π/4~π/2 : 
			1. k = (1,+∞)
		3. π/2~3π/4 :  
			1. k = (-∞,-1)
		4. 3π/4~π : 
			1. k = [-1,0)


		5. π~5π/4 : 
			1. k = [0,1]
		6. 5π/4~3π/2 : 
			1. k = (-1,+∞)
		7. 3π/2~7π/4 :  
			1. k = (-∞,-1)
		8. 7π/4~2π : 
			1. k = [-1,0)
